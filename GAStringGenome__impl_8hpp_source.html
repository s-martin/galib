59// The read specialization takes in each character whether it is whitespace or
60// not and stuffs it into the genome. This is unlike the default array read.
61template <> int GA1DArrayAlleleGenome<char>::read(std::istream &is)
62{
63 unsigned int i = 0;
64 char c;
65 do
66 {
67 is.get(c);
68 if (!is.fail()) {
69 gene(i++, c);
70}
71 } while (!is.fail() && !is.eof() && i < nx);
72
73 if (is.eof() && i < nx)
74 {
75 GAErr(GA_LOC, className(), "read", GAError::UnexpectedEOF);
76 is.clear(std::ios::badbit | is.rdstate());
77 return 1;
78 }
79 return 0;
80}

82// Unlike the base array genome, here when we write out we don't put any
83// whitespace between genes. No newline at end of it all.
84template <> int GA1DArrayAlleleGenome<char>::write(std::ostream &os) const
85{
86 for (unsigned int i = 0; i < nx; i++) {
87 os << gene(i);
88}
89 return 0;
90}
