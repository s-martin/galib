6 DESCRIPTION:
7 This header defines the interface for the binary-to-decimal genome and the
8phenotype objects used for the binary-to-decimal mappings in the
9binary-to-decimal genomes.
10 The binary-to-decimal phenotype uses reference counting so that you only
11need one instance of the underlying bit-to-float mapping. Whenever you create
12a new phenotype it gets its own core. You can use the 'link' member to make
13a phenotype refer to another's core. This is what the genomes do when they
14clone themselves so that you don't end up with multiple phenotypes running
15around. The core will persist until the last phenotype is destroyed (last one
16out, please turn off the lights).
17 When you clone a phenotype, you get a competeley new one with its own core.

60class GABin2DecPhenotype
61{
62 public:
63 GABin2DecPhenotype() : core(new GABin2DecPhenotypeCore) {}
64 GABin2DecPhenotype(const GABin2DecPhenotype &p)
65 : core(new GABin2DecPhenotypeCore(*(p.core)))
66 {
67 }
68 virtual ~GABin2DecPhenotype()
69 {
70 core->cnt -= 1;
71 if (core->cnt == 0)
72 {
73 delete core;
74 }
75 }
76 GABin2DecPhenotype &operator=(const GABin2DecPhenotype &p)
77 {
78 if (&p != this)
79 {
80 *core = *(p.core);
81 }
82 return *this;
83 }
84 GABin2DecPhenotype *clone() const { return new GABin2DecPhenotype(*this); }
85 void link(GABin2DecPhenotype &p)
86 {
87 core->cnt -= 1;
88 if (core->cnt == 0)
89 {
90 delete core;
91 }
92 core = p.core;
93 core->cnt += 1;
94 }
95 void add(unsigned int nbits, float min, float max);
96 void remove(unsigned int which);
97 unsigned int size() const { return core->sz; }
98 unsigned int nPhenotypes() const { return core->n; }
99 float min(int which) const { return core->minval[which]; }
100 float max(int which) const { return core->maxval[which]; }
101 int length(int which) const { return core->nbits[which]; }
102 int offset(int which) const { return core->oset[which]; }
103 bool equal(const GABin2DecPhenotype &) const;
104
105 protected:
106 GABin2DecPhenotypeCore *core;
107};
